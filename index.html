""" Study Roadmap Planner Subjects: Bangla, English, Life Science, Physical Science, Mathematics, History, Geography Run: python study_planner.py """ from datetime import date, datetime SUBJECTS = { "Bangla": "Read the lesson, write 5 key points, practise grammar and writing.", "English": "Read aloud, learn 10 new words, practise grammar and writing.", "Life Science": "Study diagrams, make short notes, revise terms by recalling.", "Physical Science": "Learn formulas and laws, then solve numericals.", "Mathematics": "Solve problems on paper every day. Do not just read solutions.", "History": "Make a timeline, note causes and effects, write short answers.", "Geography": "Practise maps, learn definitions, draw diagrams.", } # Harder / practice-heavy subjects get studied earlier in the day. PRIORITY = ["Mathematics", "Physical Science", "Life Science", "English", "Bangla", "Geography", "History"] DAYS = ["Monday", "Tuesday", "Wednesday", "Thursday", "Friday", "Saturday"] def ask_float(prompt, default): raw = input(f"{prompt} [{default}]: ").strip() try: return float(raw) if raw else default except ValueError: return default def pick_weak_subjects(): print("\nSubjects:", ", ".join(SUBJECTS)) raw = input("Weak subjects (comma separated, or press Enter to skip): ") weak = [s.strip().lower() for s in raw.split(",") if s.strip()] return [s for s in SUBJECTS if s.lower() in weak] def build_week(hours_per_day, weak): # Each day gets 2 subjects (1 subject if hours < 2). Weak subjects # appear more often, so they are added extra times to the queue. per_day = 1 if hours_per_day < 2 else 2 queue = list(PRIORITY) + weak # weak subjects appear twice plan, idx = {}, 0 for day in DAYS: todays = [] while len(todays) < per_day: subject = queue[idx % len(queue)] idx += 1 if subject not in todays: todays.append(subject) plan[day] = todays return plan, per_day def fmt_time(minutes): h, m = divmod(int(minutes), 60) return f"{h:02d}:{m:02d}" def print_plan(plan, per_day, hours_per_day, start_hour): total_min = hours_per_day * 60 review = 15 # nightly recap block = (total_min - review) / per_day print("\n" + "=" * 52) print(" YOUR WEEKLY STUDY ROADMAP") print("=" * 52) for day, subjects in plan.items(): print(f"\n{day}") t = start_hour * 60 for s in subjects: end = t + block print(f" {fmt_time(t)}-{fmt_time(end)} {s}") print(f" Tip: {SUBJECTS[s]}") t = end print(f" {fmt_time(t)}-{fmt_time(t + review)} Recap (close books, recall)") print("\nSunday") print(" Weekly review of all 7 subjects, fix weak topics, then rest.") def print_rules(): print("\n" + "-" * 52) print("HOW TO STUDY EACH BLOCK") print("-" * 52) print("1. Skim headings (5 min)") print("2. Study actively and write short notes") print("3. Take a 5-10 min break after every 40-50 min") print("4. Close the book and recall what you learned") print("5. Solve questions / write answers") print("Sleep 7-8 hours. Keep your phone away while studying.") def exam_countdown(): raw = input("\nExam date (YYYY-MM-DD, or Enter to skip): ").strip() if not raw: return try: exam = datetime.strptime(raw, "%Y-%m-%d").date() except ValueError: print("Date format not understood, skipping countdown.") return days_left = (exam - date.today()).days if days_left < 0: print("That date has already passed.") return print(f"\n{days_left} days left (about {days_left // 7} weeks).") if days_left > 30: print("Phase 1: finish syllabus. Phase 2 (last 30 days): revise and " "practise past papers. Last week: only revision.") else: print("Focus on revision and past papers. Do not start new topics.") def main(): print("STUDY ROADMAP PLANNER") hours = ask_float("Study hours per day", 3.0) start = ask_float("Start time (24h, e.g. 17 for 5 PM)", 17.0) weak = pick_weak_subjects() plan, per_day = build_week(hours, weak) print_plan(plan, per_day, hours, start) print_rules() exam_countdown() if __name__ == "__main__": main()
